function map(
  I: interface, O: interface,
  s: integer, op: I => O
) i: I[s] => o: O[s] {
  i[s-1] => op => o[s-1];
  if (s > 1) {
    i[0:s-2] => map(I, O, s-1, op) => o[0:s-2];
  }
